#include <Wire.h>
#include <PN532_I2C.h>
#include <PN532.h>

PN532_I2C pn532i2c(Wire);
PN532 nfc(pn532i2c);

String tagId = "None", dispTag = "None";
byte nuidPICC[4];

// Updated tag IDs with the ones you provided
String tagId1 = "117.74.133.42";  // Bharathwaj's card
String tagId2 = "90.111.220.115"; // Gowtham's card

int ledpin1 = D1;
int ledpin2 = D2;

void setup(void) {
  Serial.begin(115200);
  pinMode(ledpin1, OUTPUT);
  pinMode(ledpin2, OUTPUT);
  
  Serial.println("RFID Attendance System Started!");
  nfc.begin();
  
  uint32_t versiondata = nfc.getFirmwareVersion();
  if (!versiondata) {
    Serial.print("Didn't Find PN53x Module");
    while (1);
  }
  
  Serial.print("Found chip PN5");
  Serial.println((versiondata >> 24) & 0xFF, HEX);
  Serial.print("Firmware ver. ");
  Serial.print((versiondata >> 16) & 0xFF, DEC);
  Serial.print('.');
  Serial.println((versiondata >> 8) & 0xFF, DEC);
  
  nfc.SAMConfig();
  digitalWrite(ledpin1, LOW);
  digitalWrite(ledpin2, LOW);
}

void loop() {
  readNFC();
  
  // Check for Bharathwaj's card
  if (tagId == tagId1) {
    Serial.println("Bharathwaj present");
    if (digitalRead(ledpin1) == 0) {
      digitalWrite(ledpin1, HIGH);
      tagId = "";
      delay(1000);
    } else if (digitalRead(ledpin1) == 1) {
      digitalWrite(ledpin1, LOW);
      tagId = "";
      delay(1000);
    }
  }
  
  // Check for Gowtham's card
  if (tagId == tagId2) {
    Serial.println("Gowtham present");
    if (digitalRead(ledpin2) == 0) {
      digitalWrite(ledpin2, HIGH);
      tagId = "";
      delay(1000);
    } else if (digitalRead(ledpin2) == 1) {
      digitalWrite(ledpin2, LOW);
      tagId = "";
      delay(1000);
    }
  }
}

void readNFC() {
  boolean success;
  uint8_t uid[] = { 0, 0, 0, 0, 0, 0, 0 };
  uint8_t uidLength;
  
  success = nfc.readPassiveTargetID(PN532_MIFARE_ISO14443A, &uid[0], &uidLength);
  
  if (success) {
    Serial.print("Card detected - UID Length: ");
    Serial.print(uidLength, DEC);
    Serial.println(" bytes");
    Serial.print("UID Value: ");
    
    for (uint8_t i = 0; i < uidLength; i++) {
      nuidPICC[i] = uid[i];
      Serial.print(" "); Serial.print(uid[i], DEC);
    }
    Serial.println();
    
    tagId = tagToString(nuidPICC);
    dispTag = tagId;
    Serial.print("Card ID: ");
    Serial.println(tagId);
    Serial.println("");
    delay(1000);
  } else {
    Serial.println("Waiting for card...");
  }
}

String tagToString(byte id[4]) {
  String tagId = "";
  for (byte i = 0; i < 4; i++) {
    if (i < 3) tagId += String(id[i]) + ".";
    else tagId += String(id[i]);
  }
  return tagId;
}
