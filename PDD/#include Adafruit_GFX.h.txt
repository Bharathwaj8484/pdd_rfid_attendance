  delay(5000);
}

void readNFC() {
  if (nfc.tagPresent()) {
    NfcTag tag = nfc.read();
    tag.print();
    tagId = tag.getUidString();
    Serial.println("Tag id");
    Serial.println(tagId);
  }
  delay(1000);
}

// Function to send SMS
void sendSMS(String phoneNumber, String message) {
  sim800.print("AT+CMGS=\""); // Command to send SMS
  sim800.print(phoneNumber);
  sim800.print("\"\r");
  delay(100);
  sim800.print(message); // SMS message content
  delay(100);
  sim800.write(26); // ASCII code of CTRL+Z to send the SMS
  delay(1000);
  Serial.println("SMS sent: " + message);
}

// Function to send AT commands and check responses
void sendATCommand(String command, int waitTime) {
  sim800.println(command);
  delay(waitTime);
  while (sim800.available()) {
    String response = sim800.readString();
    Serial.println(response);
  }
}
